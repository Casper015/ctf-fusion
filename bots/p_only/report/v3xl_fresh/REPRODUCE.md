# Reproducing the T0022 supplement

The official report is generated by `python bots/p_only/tools/run_screen.py --config profiles/screen_v3xl_fresh.json` under the task CPU lock. Keep the original ignored `data/p_only/v3xl_fresh/` and `build/p_only/` artifacts; do not rerun consumed seed 20261240 as new evidence.

Extract the Python fence below into a temporary file outside the tracked source tree. From the frozen repository root, run `python <script> --snapshot <snapshot.json> --checkpoint <original-checkpoint-dir>`. The snapshot must contain the host_snapshot object from PROVENANCE.json; the checkpoint directory must contain the exact model.npz and best.json named there. Set AI_TEST_ROOT only in the environment. No absolute user-home path is needed in this committed document.

The generator rejects duplicate/stale/missing rows and incompatible hashes, checks all candidate binaries and dependencies against the frozen manifest, audits kept-match timing and counter coverage, and uses the existing analyzer's equal-count-weighted map-clustered 95% intervals. First-turn quantiles summarize first replies; subsequent quantiles summarize per-game reply statistics. It writes the compact supplement, provenance and copied model manifests. JSON retains all per-count results and all counter sums. A missing counter is not converted to a measured zero.

python <extracted-script> --snapshot <host-snapshot.json> [--checkpoint <run-dir>]
The source is embedded verbatim in report/REPRODUCE.md; no result is edited.

def read(p):
    return json.loads(Path(p).read_text(encoding='utf-8-sig'))

def sha(p):
    h = hashlib.sha256()
    with Path(p).open('rb') as f:
        for block in iter(lambda: f.read(1048576), b''):
            h.update(block)
    return h.hexdigest()

def dump(p, obj):
    Path(p).write_text(json.dumps(obj, indent=1, sort_keys=True) + '\n', encoding='utf-8')

assert len(fixtures) == 700 and cfg['seed'] == 20261240

    assert arm in rows and fid in fixtures
    assert fid not in rows[arm], ('duplicate', arm, fid)
    assert r['identity'] == ids[arm]['digest'] and r['schedule'] == sid
    assert r['source_sha256'] == ids[arm]['components']['binary_sha256']
    assert r['engine_sha256'] == '07e43f6210c640d87a9161193a71c394fde157f816af004d37a68ecc1e36a14a'
    assert r['teams'] == fixtures[fid]['teams']
    rows[arm][fid] = r
for arm in arms:
    assert set(rows[arm]) == set(fixtures)

assert ids[arms[-1]] == ids[cfg['aa'][0]]

terms = [(x+' - '+y, [(1,x),(-1,y)]) for x,y in cfg['comparisons']]
for name, expr in cfg['contrasts'].items():
    _, parsed = a.parse_contrast(name+'='+expr)

terms += [(x+'~aa - '+x, [(1,x+'~aa'),(-1,x)]) for x in cfg['aa']]
subsets = {}
for label, counts in [('2', [2]), ('3-15', list(range(3,16)))]:
    fx = {fid:f for fid,f in fixtures.items() if f['teams'] in counts}

    per_count[str(n)] = {name:a.contrast([(w,rows[arm]) for w,arm in t], fx, [n])['overall'] for name,t in terms}

# Each reply statistic is per game. Quantiles below summarize those per-game
# statistics; they are never pooled across individual turns.
timing = {arm:[] for arm in arms}
coverage = {arm:defaultdict(lambda: {'games':0,'turns':0,'games_by_count':Counter(),'turns_by_count':Counter(),'counts':defaultdict(float),'settings':{}}) for arm in arms}
game_counts = {arm:Counter() for arm in arms}
total_turns = Counter()
seen = set()
keys = {hashlib.sha256(arm.encode()).hexdigest()[:10]:arm for arm in arms}
for p in sorted((work/'matches').rglob('*.json.gz')):
    arm = keys.get(p.name.split('-')[-1].split('.')[0])
    assert arm, ('unrecognized match', p.name)
    m = json.loads(gzip.decompress(p.read_bytes()))
    bots = [b for seat,b in zip(m['seats'],m['bots']) if seat['name'].startswith('@candidate')]
    assert len(bots) == 1
    bot = bots[0]
    token = p.name.rsplit('-',1)[0], arm
    assert token not in seen
    seen.add(token)
    turns = len(m.get('blocks', []))
    n = len(m['seats'])
    fid = token[0]
    assert fid in rows[arm]
    r = rows[arm][fid]
    assert m['engine_sha256'] == r['engine_sha256']
    assert m['map_seed'] == r['map_seed'] and m['teams'] == n == r['teams']
    assert m['limits_ms'] == {'first':2000,'turn':50} or m['limits_ms'] == [2000,50] or m['limits_ms'] == {'first_ms':2000,'turn_ms':50}, m['limits_ms']
    for seat,b in zip(m['seats'],m['bots']):
        if seat['name'].startswith('@candidate'):
            assert seat['sha256'] == r['source_sha256']
    game_counts[arm][n] += 1
    total_turns[arm] += turns
    timing[arm].append({k:bot.get(k) for k in ('first_ms','p50_ms','p99_ms','max_ms','faults','late','dead','turns_timed')})
    prefixes = set()
    for line in bot.get('stderr_tail', []):
        match = a.LINE.match(line.strip())
        if not match:
            continue
        prefix = match.group(1)
        assert prefix not in prefixes, ('duplicate counter line', p.name, prefix)
        prefixes.add(prefix)
        c = coverage[arm][prefix]
        c['games'] += 1
        c['turns'] += turns
        c['games_by_count'][n] += 1
        c['turns_by_count'][n] += turns
        for k,v in a.PAIR.findall(match.group(2)):
            value = float(v)
            if k.startswith('coef_') or k in a.SETTINGS:
                if k in c['settings']:
                    assert c['settings'][k] == value
                c['settings'][k] = value
            else:
                c['counts'][k] += value
for arm in arms:
    assert game_counts[arm] == Counter({n:50 for n in cfg['teams']})
    assert len(timing[arm]) == 700

def quantiles(v):
    values = sorted(x for x in v if x is not None)
    if not values:
        return {'covered_games':0,'median':None,'p99':None,'max':None}
    return {'covered_games':len(values),'median':statistics.median(values),
            'p99':values[min(len(values)-1, int(.99*len(values)))], 'max':values[-1]}

summary = {}
for arm in arms:
    ts = timing[arm]
    cdata = {prefix:dict(c) for prefix,c in coverage[arm].items()}
    for prefix,c in cdata.items():
        c['missing_games'] = 700-c['games']
        c['total_game_turns'] = total_turns[arm]
        c['missing_turns'] = total_turns[arm]-c['turns']
    def rate(prefix, field, scale=1):
        c = cdata.get(prefix)
        return scale*c['counts'][field]/c['turns'] if c and field in c['counts'] and c['turns'] else None
    summary[arm] = {
        'games':700, 'first_ms':quantiles(t['first_ms'] for t in ts),
        'game_p50_ms':quantiles(t['p50_ms'] for t in ts),
        'game_p99_ms':quantiles(t['p99_ms'] for t in ts),
        'game_max_ms':quantiles(t['max_ms'] for t in ts),
        'faults':sum(t['faults'] for t in ts), 'late':sum(t['late'] or 0 for t in ts),
        'dead_games':sum(bool(t['dead']) for t in ts), 'counter_coverage':cdata,
        'rates_on_covered_turns':{'value_ms':rate('VALUE','seconds',1000),
            'search_samples':rate('SEARCH','samples'),
            'value_refine_calls':rate('VALUE','refine_calls'),
            'value_beam_calls':rate('VALUE','beam_calls')},
    }
    assert summary[arm]['faults'] == sum(r['faults'] for r in rows[arm].values())
    assert summary[arm]['late'] == sum(r.get('late') or 0 for r in rows[arm].values())

# Verify every candidate binary, bundle, and recorded inlined dependency after
# the run. Shared V is also checked explicitly because identity.models is empty
# for wrappers with pre-included registries.
checked = {}
for arm in cfg['arms']:
    v = manifest['variants'][arm]

        # build.py hashes the UTF-8 bundle string before Windows write_text
        # translates LF to CRLF. Reproduce that canonical source identity.
        actual = hashlib.sha256(path.read_text(encoding='utf-8').encode('utf-8')).hexdigest() if field=='bundle' else sha(path)
        assert actual == v[field+'_sha256'] == ids[arm]['components'][field+'_sha256']
        checked[path.as_posix()] = actual
        if field=='bundle':
            checked[path.as_posix()+':on_disk_bytes'] = sha(path)
    for inp in v['inputs']:
        path = Path(inp['path'])
        actual = sha(path)
        assert actual == inp['sha256'], path
        checked[path.as_posix()] = actual
model_manifests = {}
for tier, expected in [('models_pxxl_vm','6a5d24a4b5b072f30062e290457b044e09022a194edcc325c952c6962b2530ba'),
                       ('models_pxxl_v3xl','9b7ba9c2b054715e00b6f7825e6a6d5d4709f3588d37e502f4dcfec5b88f3a7a')]:

            path = directory/sv['header']

            # Export on Windows recorded the CRLF file hash; Git stores LF.
            crlf = hashlib.sha256(path.read_text(encoding='utf-8').replace('\n','\r\n').encode('utf-8')).hexdigest()
            assert sv['sha256'] in (actual,crlf), path
            checked[path.as_posix()] = actual
    dump(out/(tier+'_MANIFEST.json'),mm)
